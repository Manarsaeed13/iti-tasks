  CREATE table muiscian (
 id INT PRIMARY KEY AUTO_INCREMENT,

 //2
 CREATE TABLE musican_phone(
    muiscian_id Int NOT NULL,
     phone Varchar(20) NOT NULL,
     PRIMARY KEY( muiscian_id,phone),
      FOREIGN KEY ( muiscian_id) REFERENCES muiscian(id)
        ON DELETE CASCADE ON UPDATE CASCADE

 );


   //3
    CREATE TABLE album(
    id INT PRIMARY KEY AUTO_INCREMENT,
    title varchar(255) NOT NULL,
    cr_date DATE,
    muiscian_id INT NOT NULL,
    FOREIGN KEY ( muiscian_id)REFERENCES muiscian(id) 
    ON DELETE  CASCADE ON UPDATE CASCADE
    
    

);


//4
CREATE TABLE instrament(
       Name VARCHAR(100) PRIMARY KEY,
     `Key` VARCHAR(50)
    

);



//5
CREATE TABLE song (
title varchar(255) PRIMARY KEY,
 author VARCHAR(100),
  album_id INT NOT NULL,
   FOREIGN KEY (  album_id)REFERENCES  album(id)
   ON DELETE  CASCADE ON UPDATE CASCADE
    
 
);


//6
CREATE TABLE play_instrument (
  musician_id INT NOT NULL,
    instrument_Name VARCHAR(100) NOT NULL,
    PRIMARY KEY( musician_id, instrument_Name),
    
     FOREIGN KEY (musician_id) REFERENCES muiscian(id)
     ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (instrument_Name) REFERENCES   instrament(Name) ON DELETE CASCADE ON UPDATE CASCADE      
      


);


//7
CREATE TABLE play_song (
    musician_id INT NOT NULL,
    song_title VARCHAR(255) NOT NULL,
    
     PRIMARY KEY (musician_id, song_title),
      FOREIGN KEY (musician_id) REFERENCES muiscian(id)
     ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (song_title) REFERENCES song(title) 
    ON DELETE CASCADE ON UPDATE CASCADE      
      
);

 create database task_sales;

 //1

CREATE TABLE sales_office (
    NUM INT PRIMARY KEY AUTO_INCREMENT,
    Loc VARCHAR(255),
    employee_ID INT UNIQUE,  
    FOREIGN KEY (employee_ID) REFERENCES employee(ID)
    ON DELETE CASCADE ON UPDATE CASCADE      
);


//2

CREATE TABLE employee (
    ID INT PRIMARY KEY,
    Name VARCHAR(100),
    Office_NUM INT,
    CONSTRAINT fk_employee_office FOREIGN KEY (Office_NUM) REFERENCES sales_office(NUM)
);


//3

CREATE TABLE property (
    ID INT PRIMARY KEY AUTO_INCREMENT,
    Add_ VARCHAR(255),
    City VARCHAR(100),
    State VARCHAR(100),
    Zip VARCHAR(20),
    office_num_fk INT NOT NULL,

    FOREIGN KEY (office_num_fk) REFERENCES sales_office(NUM)
        ON DELETE CASCADE ON UPDATE CASCADE
);


//4
CREATE TABLE owner (
    ID INT PRIMARY KEY AUTO_INCREMENT,
    Name VARCHAR(200) NOT NULL
);


//5
CREATE TABLE property_owner (
    property_id INT NOT NULL,
    owner_id INT NOT NULL,
    percent_owned DECIMAL(5,2),  

    PRIMARY KEY (property_id, owner_id),

    FOREIGN KEY (property_id) REFERENCES property(ID)
        ON DELETE CASCADE ON UPDATE CASCADE,
    FOREIGN KEY (owner_id) REFERENCES owner(ID)
        ON DELETE CASCADE ON UPDATE CASCADE
);
